369template <SupportedFigure T>
370T* GeometryStorage::get(ID id) noexcept {
371 auto it = m_index.find(id);
372 if (it == m_index.end() || it->second.type != typeToEnum<T>()) {
373 return nullptr;
374 }
375 const std::uint32_t s = it->second.slot;
376 if constexpr (std::same_as<T, Point2D>) {
377 return m_pointSlots[s].get();
378 } else if constexpr (std::same_as<T, Line2D>) {
379 return m_lineSlots[s].get();
380 } else if constexpr (std::same_as<T, Circle2D>) {
381 return m_circleSlots[s].get();
382 } else {
383 return m_arcSlots[s].get();
384 }
385}

409template <typename T>
410void GeometryStorage::eraseCachedById(
411 ID id,
412 std::vector<FigureRef<T>>& cache,
413 std::unordered_map<ID, std::size_t>& posMap
414) noexcept {
415 auto it = posMap.find(id);
416 if (it == posMap.end()) {
417 return;
418 }
419 const std::size_t idx = it->second;
420 const std::size_t lastIdx = cache.size() - 1;
421 if (idx != lastIdx) {
422 cache[idx] = cache[lastIdx];
423 posMap[cache[idx].id] = idx;
424 }
425 cache.pop_back();
426 posMap.erase(it);
427}
