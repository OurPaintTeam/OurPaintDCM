393template <SupportedFigure T>
394T* GeometryStorage::get(ID id) noexcept {
395 auto it = m_index.find(id);
396 if (it == m_index.end() || it->second.type != typeToEnum<T>()) {
397 return nullptr;
398 }
399 const std::uint32_t s = it->second.slot;
400 if constexpr (std::same_as<T, Point2D>) {
401 return m_pointSlots[s].get();
402 } else if constexpr (std::same_as<T, Line2D>) {
403 return m_lineSlots[s].get();
404 } else if constexpr (std::same_as<T, Circle2D>) {
405 return m_circleSlots[s].get();
406 } else if constexpr (std::same_as<T, Arc2D>) {
407 return m_arcSlots[s].get();
408 } else {
409 return m_curveSlots[s].get();
410 }
411}

437template <typename T>
438void GeometryStorage::eraseCachedById(
439 ID id,
440 std::vector<FigureRef<T>>& cache,
441 std::unordered_map<ID, std::size_t>& posMap
442) noexcept {
443 auto it = posMap.find(id);
444 if (it == posMap.end()) {
445 return;
446 }
447 const std::size_t idx = it->second;
448 const std::size_t lastIdx = cache.size() - 1;
449 if (idx != lastIdx) {
450 cache[idx] = cache[lastIdx];
451 posMap[cache[idx].id] = idx;
452 }
453 cache.pop_back();
454 posMap.erase(it);
455}
