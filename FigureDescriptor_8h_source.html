164 bool validate() const {
165 requireFiniteValues(coords);
166 requireFinite(x);
167 requireFinite(y);
168 if (radius) requirePositiveRadius(*radius);
169 switch (type) {
170 case FigureType::ET_POINT2D:
171 if (!coords.empty() && coords.size() != 2) {
172 throw std::invalid_argument("Point requires exactly 2 coordinates");
173 }
174 if (coords.size() == 2) {
175 break;
176 }
177 if (!x.has_value() || !y.has_value()) {
178 throw std::invalid_argument("Point requires x and y coordinates");
179 }
180 break;
181 case FigureType::ET_LINE:
182 if (!coords.empty() && coords.size() != 4) {
183 throw std::invalid_argument("Line requires exactly 4 coordinates");
184 }
185 if (coords.size() == 4) {
186 break;
187 }
188 if (pointIds.size() != 2) {
189 throw std::invalid_argument("Line requires coordinates or exactly 2 point IDs");
190 }
191 break;
192 case FigureType::ET_CIRCLE:
193 if (!coords.empty() && coords.size() != 2) {
194 throw std::invalid_argument("Circle requires exactly 2 center coordinates");
195 }
196 if (coords.size() != 2 && pointIds.size() != 1) {
197 throw std::invalid_argument("Circle requires center coordinates or exactly 1 center point ID");
198 }
199 if (!radius.has_value() || radius.value() <= 0) {
200 throw std::invalid_argument("Circle requires positive radius");
201 }
202 break;
203 case FigureType::ET_ARC:
204 if (!coords.empty() && coords.size() != 6) {
205 throw std::invalid_argument("Arc requires exactly 6 coordinates");
206 }
207 if (coords.size() == 6) {
208 break;
209 }
210 if (pointIds.size() != 3) {
211 throw std::invalid_argument("Arc requires coordinates or exactly 3 point IDs");
212 }
213 break;
214 case FigureType::ET_CURVE2D:
215 if (coords.size() % 2 != 0) {
216 throw std::invalid_argument("Curve controls require x,y coordinate pairs");
217 }
218 Figures::Curve2D::validateData(degree,
219 coords.empty() ? pointIds.size() : coords.size() / 2, weights, knots, periodic);
220 break;
221 default:
222 throw std::invalid_argument("Unsupported figure type");
223 }
224 return true;
225 }
