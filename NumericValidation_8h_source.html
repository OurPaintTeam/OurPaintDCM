43inline void validateRequirementParameter(RequirementType type, const std::optional<double>& param) {
44 switch (type) {
45 case RequirementType::ET_ARCARCTANGENT:
46 case RequirementType::ET_CIRCLECIRCLETANGENT:
47 if (!param || (*param != 0 && *param != 1 && *param != 2))
48 throw std::invalid_argument("Circle tangency kind must be 0, 1 or 2");
49 break;
50 case RequirementType::ET_ARCLINETANGENT:
51 case RequirementType::ET_LINECIRCLETANGENT:
52 if (!param || (*param != -1 && *param != 1))
53 throw std::invalid_argument("Tangency side must be -1 or +1");
54 break;
55 case RequirementType::ET_SYMMETRICABOUTHORIZONTAL:
56 case RequirementType::ET_SYMMETRICABOUTVERTICAL:
57 if (!param) throw std::invalid_argument("Axis symmetry needs an axis coordinate");
58 requireFinite(*param);
59 break;
60 case RequirementType::ET_CIRCLERADIUS:
61 case RequirementType::ET_ARCRADIUS:
62 case RequirementType::ET_CIRCLEDIAMETER:
63 if (!param) throw std::invalid_argument("Size requirement needs a parameter");
64 requirePositiveRadius(*param);
65 if (type == RequirementType::ET_CIRCLEDIAMETER) requirePositiveRadius(*param / 2.0);
66 break;
67 case RequirementType::ET_POINTCIRCLEDIST:
68 case RequirementType::ET_POINTARCDIST:
69 if (!param) throw std::invalid_argument("Point-circle/arc distance needs a parameter");
70 requireFinite(*param);
71 break;
72 case RequirementType::ET_POINTLINEDIST:
73 case RequirementType::ET_POINTPOINTDIST:
74 case RequirementType::ET_POINTPOINTHORIZONTALDIST:
75 case RequirementType::ET_POINTPOINTVERTICALDIST:
76 case RequirementType::ET_LINECIRCLEDIST:
77 case RequirementType::ET_LINEARCDIST:
78 if (!param) throw std::invalid_argument("Distance requirement needs a parameter");
79 requireNonNegative(*param);
80 break;
81 case RequirementType::ET_ARCANGLE:
82 if (!param) throw std::invalid_argument("Arc angle needs a parameter");
83 requireFinite(*param);

92 }
93 break;
