285 bool validate() const {
286 if (!std::isfinite(weight) || weight < 0.0) {
287 throw std::invalid_argument("Requirement weight must be finite and non-negative");
288 }
289 if (weight != 1.0 && weight != 0.0 &&
290 (type == RequirementType::ET_POINTONPOINT ||
291 type == RequirementType::ET_FIXPOINT ||
292 type == RequirementType::ET_FIXLINE ||
293 type == RequirementType::ET_FIXARC ||
294 type == RequirementType::ET_FIXCIRCLE)) {
295 throw std::invalid_argument("Eliminated and fixed requirements must use weight 0 or 1");
296 }
297 switch (type) {
298 case RequirementType::ET_SYMMETRICABOUTLINE:
299 case RequirementType::ET_SYMMETRICABOUTPOINT:
300 if (objectIds.size() != 3)
301 throw std::invalid_argument("Symmetry requires exactly 3 object IDs");
302 break;
303 case RequirementType::ET_POINTLINEDIST:
304 case RequirementType::ET_POINTONLINE:
305 case RequirementType::ET_POINTPOINTDIST:
306 case RequirementType::ET_POINTPOINTHORIZONTALDIST:
307 case RequirementType::ET_POINTPOINTVERTICALDIST:
308 case RequirementType::ET_POINTPOINTHORIZONTAL:
309 case RequirementType::ET_POINTPOINTVERTICAL:
310 case RequirementType::ET_POINTONPOINT:
311 case RequirementType::ET_LINECIRCLEDIST:
312 case RequirementType::ET_LINEARCDIST:
313 case RequirementType::ET_LINEONCIRCLE:
314 case RequirementType::ET_LINEINCIRCLE:
315 case RequirementType::ET_LINELINEPARALLEL:
316 case RequirementType::ET_LINELINEPERPENDICULAR:
317 case RequirementType::ET_LINELINEANGLE:
318 case RequirementType::ET_POINTONCIRCLE:
319 case RequirementType::ET_POINTCIRCLEDIST:
320 case RequirementType::ET_POINTONARC:
321 case RequirementType::ET_POINTARCDIST:
322 case RequirementType::ET_EQUALLENGTH:
323 case RequirementType::ET_EQUALRADIUS:
324 case RequirementType::ET_POINTATMIDPOINT:
325 case RequirementType::ET_LINELINEMIDPOINT:
326 case RequirementType::ET_ARCARCMIDPOINT:
327 case RequirementType::ET_LINEARCMIDPOINT:
328 case RequirementType::ET_POINTATARCMIDPOINT:
329 case RequirementType::ET_LINECIRCLETANGENT:
330 case RequirementType::ET_CIRCLECIRCLETANGENT:
331 case RequirementType::ET_ARCLINETANGENT:
332 case RequirementType::ET_ARCARCTANGENT:
333 case RequirementType::ET_SYMMETRICABOUTHORIZONTAL:
334 case RequirementType::ET_SYMMETRICABOUTVERTICAL:
335 if (objectIds.size() != 2) {
336 throw std::invalid_argument("Requirement type requires exactly 2 object IDs");
337 }
338 break;
339 case RequirementType::ET_VERTICAL:
340 case RequirementType::ET_HORIZONTAL:
341 case RequirementType::ET_ARCCENTERONPERPENDICULAR:
342 case RequirementType::ET_FIXPOINT:
343 case RequirementType::ET_FIXLINE:
344 case RequirementType::ET_FIXARC:
345 case RequirementType::ET_FIXCIRCLE:
346 case RequirementType::ET_CIRCLERADIUS:
347 case RequirementType::ET_ARCRADIUS:
348 case RequirementType::ET_ARCANGLE:
349 case RequirementType::ET_CIRCLEDIAMETER:
350 if (objectIds.size() != 1) {
351 throw std::invalid_argument("Requirement type requires exactly 1 object ID");
352 }
353 break;
354 default:
355 throw std::invalid_argument("Unsupported requirement type");
356 }
357 validateRequirementParameter(type, param);
358 return true;
359 }
